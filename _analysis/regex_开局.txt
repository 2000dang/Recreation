<!DOCTYPE html>
<html lang="zh-CN">
<head>
  <meta charset="UTF-8" />
  <meta name="viewport" content="width=device-width, initial-scale=1.0" />
  <title>opening shell</title>
  <style>
    html, body {
      margin: 0;
      padding: 0;
      background: transparent;
    }
  </style>
</head>
<body>
<script>
(function () {
  var url = 'https://cdn.jsdelivr.net/gh/Unspoken-MomoTea/Battlefield-of-Reincarnation@main/dist/V20260719/开局.html?v=1';

  function injectEmbedFix(html) {
    var style = [
      '<style id="dnf-shell-embed-fix">',
      'html {',
      '  height: auto !important;',
      '  min-height: 0 !important;',
      '}',
      'body {',
      '  height: auto !important;',
      '  min-height: 0 !important;',
      '}',
      '</style>'
    ].join('\n');

    var script = [
      '<script>',
      '(function () {',
      '  var raf = 0;',
      '  var ro = null;',
      '  var mo = null;',
      '',
      '  function getHeight() {',
      '    var app = document.querySelector(".wizard-layout") || document.body',
      '    if (!app) return 600;',
      '',
      '    var body = document.body;',
      '    var bodyStyle = body ? getComputedStyle(body) : null;',
      '    var pt = bodyStyle ? (parseFloat(bodyStyle.paddingTop) || 0) : 0;',
      '    var pb = bodyStyle ? (parseFloat(bodyStyle.paddingBottom) || 0) : 0;',
      '',
      '    var rect = app.getBoundingClientRect();',
      '    var appHeight = Math.max(',
      '      app.scrollHeight || 0,',
      '      app.offsetHeight || 0,',
      '      rect.height || 0',
      '    );',
      '',
      '    return Math.max(1, Math.ceil(appHeight + pt + pb));',
      '  }',
      '',
      '  function applyHeight() {',
      '    raf = 0;',
      '    var h = getHeight();',
      '    try {',
      '      if (window.frameElement) {',
      '        window.frameElement.style.minHeight = "0px";',
      '        window.frameElement.style.height = h + "px";',
      '      }',
      '    } catch (e) {}',
      '  }',
      '',
      '  function queue() {',
      '    if (raf) return;',
      '    raf = requestAnimationFrame(applyHeight);',
      '  }',
      '',
      '  function install() {',
      '    var app = document.querySelector(".wizard-layout") || document.body;',
      '    if (!app) {',
      '      setTimeout(queue, 100);',
      '      setTimeout(queue, 300);',
      '      setTimeout(queue, 800);',
      '      return;',
      '    }',
      '',
      '    if (window.ResizeObserver) {',
      '      ro = new ResizeObserver(queue);',
      '      ro.observe(app);',
      '    }',
      '',
      '    if (window.MutationObserver) {',
      '      mo = new MutationObserver(queue);',
      '      mo.observe(app, {',
      '        childList: true,',
      '        subtree: true,',
      '        attributes: true',
      '      });',
      '    }',
      '',
      '    window.addEventListener("load", queue);',
      '    window.addEventListener("resize", queue);',
      '',
      '    if (document.fonts && document.fonts.ready) {',
      '      document.fonts.ready.then(queue).catch(function () {});',
      '    }',
      '',
      '    Array.prototype.forEach.call(document.images || [], function (img) {',
      '      if (!img.complete) {',
      '        img.addEventListener("load", queue, { once: true });',
      '        img.addEventListener("error", queue, { once: true });',
      '      }',
      '    });',
      '',
      '    setTimeout(queue, 0);',
      '    setTimeout(queue, 50);',
      '    setTimeout(queue, 150);',
      '    setTimeout(queue, 400);',
      '    setTimeout(queue, 1000);',
      '  }',
      '',
      '  if (document.readyState === "loading") {',
      '    document.addEventListener("DOMContentLoaded", install, { once: true });',
      '  } else {',
      '    install();',
      '  }',
      '})();',
      '<\/script>'
    ].join('\n');

    if (html.indexOf('</head>') !== -1) {
      html = html.replace('</head>', style + '\n</head>');
    } else {
      html = style + html;
    }

    if (html.indexOf('</body>') !== -1) {
      html = html.replace('</body>', script + '\n</body>');
    } else {
      html += script;
    }

    return html;
  }

  fetch(url, { cache: 'no-store' })
    .then(function (res) {
      if (!res.ok) throw new Error(String(res.status));
      return res.text();
    })
    .then(function (html) {
      document.open();
      document.write(injectEmbedFix(html));
      document.close();
    })
    .catch(function (err) {
      document.body.innerHTML = '<pre style="padding:16px;">加载失败：' + err.message + '</pre>';
    });
})();
</script>
</body>
</html>
