        try {
            const chat_message = getChatMessages(getCurrentMessageId())[0];
            const matchContent = chat_message.message;
            // console.log('原始消息:', matchContent);

            // 解析策略: 合并所有 <mission> 块内容为文本池, 按【世界名】切分世界
            // 关键: 世界名后必紧跟 "(综合评级" (评级行首), 字段值【稳定】【扰动】【平静】等不会跟,
            //        这样可在单 <mission> 块内安全拆出多个世界
            const missionBlockRegex = /<mission>([\s\S]*?)<\/mission>/gi;
            let pool = '';
            let blockMatch;
            while ((blockMatch = missionBlockRegex.exec(matchContent)) !== null) {
                pool += '\n' + blockMatch[1];
            }
            // 回退: 若无 <mission> 标签, 用全文当文本池
            if (!pool.trim()) pool = matchContent;

            const worldData = [];
            const seenTitles = new Set();

            // 锚定世界起点: 【世界名】 后紧跟 (\s*综合评级 (评级行首部)
            // 新格式世界名前带 "1. 2. " 序号, 序号一并计入匹配,
            // 使下一个世界的起点落在其序号开头, 避免序号残留到上一世界描述末尾 (如 "...[预期收益]: xxx\n2.")
            // 第二捕获组用于回算 titleEnd (保留 "(" 在描述里, 不被吞掉)
            const worldStartRegex = /(?:^|\n)[ \t]*(?:\d+\.[ \t]*)?【([^】]+)】(\s*\(\s*综合评级)/g;
            const starts = [];
            let ws;
            while ((ws = worldStartRegex.exec(pool)) !== null) {
                // titleEnd = 【世界名】 结束位置 (不含后面评级行的空白与"(")
                const titleEnd = ws.index + (ws[0].length - ws[2].length);
                starts.push({ title: ws[1].trim(), idx: ws.index, titleEnd: titleEnd });
            }

            for (let i = 0; i < starts.length; i++) {
                const s = starts[i];
                // 描述 = 【世界名】之后(含 "(综合评级:..." 评级行), 到下一个世界起点之前
                const nextStart = (i + 1 < starts.length) ? starts[i + 1].idx : pool.length;
                const descRaw = pool.slice(s.titleEnd, nextStart).trim();
                // 过滤格式示例/占位: 描述过短或省略号过多则跳过
                const ellipsisCount = (descRaw.match(/…/g) || []).length;
                if (descRaw.length >= 20 && ellipsisCount <= 10 && !seenTitles.has(s.title)) {
                    seenTitles.add(s.title);
                    worldData.push({ title: s.title, description: descRaw });
                }
            }

            // 回退 A: 新格式无 "(综合评级" 标记时, 按 【世界名】+换行+[字段 配对
            if (worldData.length === 0) {
                const worldPatternA = /【([^】]+)】\s*\n\s*\[([\s\S]*?)(?=【[^】]+】\s*\n\s*\[|$)/g;
                let m;
                while ((m = worldPatternA.exec(pool)) !== null) {
                    const t = m[1].trim();
                    const d = m[2].trim();
                    const ec = (d.match(/…/g) || []).length;
                    if (d.length >= 50 && ec <= 5 && !seenTitles.has(t)) {
                        seenTitles.add(t);
                        worldData.push({ title: t, description: d });
                    }
                }
            }

            // 回退 B: 旧格式 【世界名】紧跟[描述] 同行配对
            if (worldData.length === 0) {
                const worldPatternB = /【([^】]+)】[\s\n]*\[([\s\S]*?)\]/g;
                let m;
                while ((m = worldPatternB.exec(pool)) !== null) {
                    const t = m[1].trim();
                    const d = m[2].trim();
                    const ec = (d.match(/…/g) || []).length;
                    if (d.length >= 50 && ec <= 5 && !seenTitles.has(t)) {
                        seenTitles.add(t);
                        worldData.push({ title: t, description: d });
                    }
                }
            }

            if (worldData.length === 0) {
                throw new Error("未找到 <mission> 标签或未能解析出世界数据");
            }

            // 动态生成世界选项HTML
            const container = document.querySelector('.options-container');
            container.innerHTML = ''; // 清空加载指示器

            worldData.forEach((world, index) => {
                const worldElement = document.createElement('div');
                worldElement.className = 'world-option';
                worldElement.setAttribute('data-index', index);

                const parsed = parseDescription(world.description);
                worldElement.innerHTML = `

        // ===== 结构化解析世界描述 =====
        // 评级危险度: 取挑战极限(范围右端)判定, F/E低危, D/C中危, B/A高危, S+极危
        function ratingTier(ratingCore) {
            const high = String(ratingCore).split('-').pop().trim().toUpperCase();
            const order = ['F', 'E', 'D', 'C', 'B', 'A', 'S', 'SS', 'SSS'];
            const idx = order.indexOf(high);
            if (idx <= 1) return 'low';
            if (idx <= 3) return 'mid';
            if (idx <= 5) return 'high';
            return 'extreme';
        }

        // 把描述文本解析为 { rating, ratingCore, tier, rows[] }
        // rows[] 每项: { label, value, parts[] }; parts 为按 | 拆分的子项
        function parseDescription(desc) {
            const result = { rating: null, ratingCore: '', tier: 'mid', rows: [] };
            const lines = String(desc).replace(/\r/g, '').split('\n');
            for (let raw of lines) {
                const line = raw.trim();
                if (!line) continue;
                // 评级行: (综合评级: E-C级)
                const rm = line.match(/^\(\s*综合评级\s*[：:]\s*([^）]+?)\s*级\s*\)\s*$/);
                if (rm) {
                    result.ratingCore = rm[1].trim();
                    result.rating = result.ratingCore + '级';
                    result.tier = ratingTier(result.ratingCore);
                    continue;
                }
                // 字段行 [label]: value 或 【label】：value
                const fm = line.match(/^(\[[^\]]+\]|【[^】]+】)\s*[：:]\s*([\s\S]*)$/);
                if (fm) {
                    const label = fm[1].replace(/[\[\]【】]/g, '').trim();
                    const value = fm[2].trim();
                    result.rows.push({ label, value, parts: value.split(/\s*\|\s*/).filter(p => p.trim()) });
                } else {
                    result.rows.push({ label: '', value: line, parts: [line] });
                }
            }
            return result;
        }

        // 识别 "模式: xxx" 子项的配色类
        function modeClass(part) {
            const mm = part.match(/模式\s*[：:]\s*(\S+)/);
            if (!mm) return '';
            const mode = mm[1];
            if (/竞争|团战|敌对/.test(mode)) return 'wd-mode-competitive';
            if (/单人|固定/.test(mode)) return 'wd-mode-solo';
            if (/多人|同场|中立/.test(mode)) return 'wd-mode-neutral';
            return '';
        }

        // 渲染结构化卡片内容
        function renderDescriptionHtml(parsed) {
            let html = '';
            if (parsed.rating) {
                html += `<div class="world-rating"><span class="rating-badge tier-${parsed.tier}">综合评级 ${escapeHtml(parsed.rating)}</span></div>`;
            }
            html += '<div class="world-fields">';
            parsed.rows.forEach(row => {
                if (!row.label) {
                    html += `<div class="wd-row wd-row-plain">${escapeHtml(row.value)}</div>`;
                    return;
                }
                const isGoal = /锚定目标|任务/.test(row.label);
                const isKey = /预期收益/.test(row.label);
                const cls = isGoal ? 'wd-row wd-row-goal' : (isKey ? 'wd-row wd-row-key' : 'wd-row');
                let valueHtml;
                if (row.parts.length > 1) {
                    valueHtml = row.parts.map((p, i) => {
                        if (i === 0) return escapeHtml(p);
                        return `<span class="wd-sep">|</span> <span class="wd-part-dim ${modeClass(p)}">${escapeHtml(p)}</span>`;
                    }).join('');
                } else {
                    valueHtml = escapeHtml(row.value);
                }
                html += `<div class="${cls}"><span class="wd-label">${escapeHtml(row.label)}</span><span class="wd-value">${valueHtml}</span></div>`;
            });
            html += '</div>';
            return html;
        }

        function escapeHtml(s) {
            if (typeof s !== 'string') return '';
            return s.replace(/&/g, '&amp;')
                .replace(/</g, '&lt;')
                .replace(/>/g, '&gt;')
                .replace(/"/g, '&quot;');
        }
        let selectedIndex = null;
        const confirmButton = document.getElementById('executeBtn');

        // 绑定世界选项事件的函数
        function bindWorldOptionEvents() {
            const worldOptions = document.querySelectorAll('.world-option');

            worldOptions.forEach(option => {
                option.addEventListener('click', function () {
                    // 移除所有选中状态
                    worldOptions.forEach(opt => opt.classList.remove('selected'));

                    // 添加当前选中状态
                    this.classList.add('selected');
                    selectedIndex = parseInt(this.getAttribute('data-index'));

                    // console.log('选中世界索引:', selectedIndex);
                });
            });
        }

        confirmButton.addEventListener('click', function () {
            if (selectedIndex === null) {
                alert('错误：未选择任何世界坐标。');
                return;
            }

            const selectedOption = document.querySelector(`.world-option[data-index="${selectedIndex}"]`);
            if (!selectedOption) {
                alert('错误：选中的世界不存在。');
                return;
            }

            confirmButton.disabled = true;
            confirmButton.textContent = 'TRANSMITTING...';

            const worldTitle = selectedOption.querySelector('.world-title').textContent.trim();
            // 发送时用 data-raw 保留原始换行 (AI 现用 [子字段]: 分行式, 不再需要 " - " 列表转换)
            const descEl = selectedOption.querySelector('.world-description');
            const worldDescription = (descEl.dataset.raw || descEl.textContent || '')
                .replace(/\r/g, '')
                .replace(/\n{3,}/g, '\n\n')   // 压缩多余空行
                .trim();

            const message = `
## 【世界信息】
**世界名称：** ${worldTitle}

## 【世界背景】
${worldDescription}

## 正在进入该世界...
      `.trim();

            // 把可确认的世界资料写入 stat_data.世界 (名称/难度/模式/异端上限与活跃余量)
            try {
                const parsed = parseDescription(worldDescription);
                const win = window.parent || window;
                if (win.Mvu && win._) {
                    const c = win.Mvu.getMvuData({type:'message', message_id:'latest'});
                    const _set = win._.set;
                    _set(c, 'stat_data.世界.名称', worldTitle);
                    if (parsed.rating) _set(c, 'stat_data.世界.难度', parsed.rating);
                    // 干涉模式行: 提取模式文本 + 预计异端数量(上限与活跃余量初始一致)
                    const modeRow = parsed.rows.find(r => /干涉模式/.test(r.label));
                    if (modeRow) {
                        const alienMatch = modeRow.value.match(/预计异端[:：]\s*(\d+)/);
                        const alienCount = alienMatch ? parseInt(alienMatch[1], 10) : 0;
                        const modeText = modeRow.value.replace(/\s*\(?\s*预计异端[:：]\s*\d+人?\s*\)?\s*/g, '').trim();
                        _set(c, 'stat_data.世界.异端雷达.当前模式', modeText);
                        _set(c, 'stat_data.世界.异端雷达.异端上限', alienCount);
                        _set(c, 'stat_data.世界.异端雷达.活跃余量', alienCount);
                    }
                    win.Mvu.replaceMvuData(c, {type:'message', message_id:'latest'});
                    try { win.Mvu.replaceMvuData(c, {type:'chat'}); } catch(e2){}
                }
            } catch(e) { console.warn('写入世界数据失败:', e); }

            sendData(message);

            setTimeout(() => {
                confirmButton.disabled = false;
                confirmButton.textContent = 'Confirm Selection';
            }, 5000);
        });

        function sendData(data) {
            // console.log('准备发送指令...');
            confirmButton.textContent = '指令已发送';

            try {
                if (typeof triggerSlash === 'function') {
                    // 转义消息内容中的管道符 |, 防止被 STScript 当作命令分隔符截断
                    // (世界描述含 "[时空/匹配]: xxx | 模式: yyy" 等管道符, 会导致 /send 在首个 | 处被截断)
                    const escapedData = data.replace(/\|/g, '\\|');
                    triggerSlash(`/send ${escapedData}|/trigger`);
                    // console.log('成功通过 triggerSlash 发送');
                    return;
                }
            } catch (e) { console.warn('triggerSlash 不可用'); }

            try {
                if (window.parent && window.parent !== window) {
                    window.parent.postMessage({ type: 'world_selection', content: data }, '*');
                    // console.log('成功通过 postMessage 发送');
                    return;
                }
            } catch (e) { console.warn('postMessage 不可用'); }

            try {
                navigator.clipboard.writeText(data).then(() => {
                    // console.log('指令已复制到剪贴板');
                    alert('指令已复制到剪贴板！请手动粘贴发送。');
                    confirmButton.textContent = '已复制到剪贴板';
                });
            } catch (e) {
                // console.error('所有发送方式均失败:', e);
                alert('自动发送失败，请手动复制指令。');
                confirmButton.textContent = '发送失败';
            }
        }

    </script>
</body>

</html>
